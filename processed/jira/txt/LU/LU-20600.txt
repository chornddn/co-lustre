--- Description ---
When an OST crosses the osp..reserved_mb_low / osp..reserved_ino_low watermark, the MDT stops allocating new objects on it, and resumes only once the corresponding *_high watermark is crossed back.  Both edges are detected in osp_pre_update_msfs(), but are only reported at

CDEBUG(D_INFO) level, which is not in the default debug mask – so the transition is effectively invisible to the administrator, who is left to poll osp.*.prealloc_status to find out the current state.

        if (msfs->os_state & (OS_STATFS_ENOINO | OS_STATFS_ENOSPC)) {
                d->opd_pre_status = -ENOSPC;
                if (!(old_state & (OS_STATFS_ENOINO | OS_STATFS_ENOSPC)))
                        CDEBUG(D_INFO, "%s: full: state=%x: rc = %x\n",
                               d->opd_obd->obd_name, msfs->os_state,
                               d->opd_pre_status);
                CDEBUG(D_INFO, "uncommitted changes=%u in_progress=%u\n",
                       atomic_read(&d->opd_sync_changes),
                       atomic_read(&d->opd_sync_rpcs_in_progress));
        } else if (old_state & (OS_STATFS_ENOINO | OS_STATFS_ENOSPC)) {
                d->opd_pre_status = 0;
                spin_lock(&d->opd_pre_lock);
                d->opd_pre_create_slow = 0;
                d->opd_pre_create_count = OST_MIN_PRECREATE;
                spin_unlock(&d->opd_pre_lock);
                wake_up(&d->opd_pre_waitq);                CDEBUG(D_INFO,
                       "%s: available: state=%x: rc = %d\n",
                       d->opd_obd->obd_name, msfs->os_state,
                       d->opd_pre_status);
        } else {
                /* we only get here if rc == 0 in the caller */
                d->opd_pre_status = 0;
        } 

Both transitions should be reported on the console, together with the free space, free inodes, and the watermarks they were compared against.
